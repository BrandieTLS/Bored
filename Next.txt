    <script>

        const table =
            document.getElementById("warrantyTable");

        const filterMenu =
            document.getElementById("filterMenu");

        const filterOptions =
            document.getElementById("filterOptions");

        const noResults =
            document.getElementById("noResults");


        /*
        ========================================
        OPEN / CLOSE FILTER
        ========================================
        */

        function toggleFilter() {

            filterMenu.classList.toggle("show");

        }


        /*
        ========================================
        CREATE FILTERS AUTOMATICALLY
        ========================================

        This reads the table headers and
        automatically creates a filter for
        every column.
        */

        function createFilters() {

            const headers =
                table.querySelectorAll("thead th");

            filterOptions.innerHTML = "";


            headers.forEach((header, columnIndex) => {

                const columnName =
                    header.textContent.trim();


                /*
                Get all unique values
                from this column.
                */

                const values = new Set();


                const rows =
                    table.querySelectorAll("tbody tr");


                rows.forEach(row => {

                    const cell =
                        row.cells[columnIndex];

                    if (cell) {

                        const value =
                            cell.textContent.trim();

                        values.add(value);

                    }

                });


                /*
                Create column section
                */

                const columnDiv =
                    document.createElement("div");

                columnDiv.className =
                    "filter-column";


                /*
                Column title
                */

                const title =
                    document.createElement("div");

                title.className =
                    "filter-column-title";

                title.textContent =
                    columnName;


                columnDiv.appendChild(title);


                /*
                Create checkbox for
                every unique value
                */

                [...values]
                    .sort()
                    .forEach(value => {

                        const label =
                            document.createElement("label");

                        label.className =
                            "filter-option";


                        const checkbox =
                            document.createElement("input");

                        checkbox.type =
                            "checkbox";

                        checkbox.dataset.column =
                            columnIndex;

                        checkbox.dataset.value =
                            value;


                        label.appendChild(checkbox);


                        const text =
                            document.createTextNode(
                                value || "(Blank)"
                            );

                        label.appendChild(text);


                        columnDiv.appendChild(label);

                    });


                filterOptions.appendChild(
                    columnDiv
                );

            });

        }


        /*
        ========================================
        APPLY FILTERS
        ========================================
        */

        function applyFilters() {

            /*
            Store selected values
            by column.
            */

            const selectedFilters = {};


            const checked =
                filterOptions.querySelectorAll(
                    "input[type='checkbox']:checked"
                );


            checked.forEach(checkbox => {

                const column =
                    checkbox.dataset.column;

                const value =
                    checkbox.dataset.value;


                if (!selectedFilters[column]) {

                    selectedFilters[column] = [];

                }


                selectedFilters[column].push(
                    value
                );

            });


            /*
            Check every table row.
            */

            const rows =
                table.querySelectorAll(
                    "tbody tr"
                );


            let visibleRows = 0;


            rows.forEach(row => {

                let showRow = true;


                /*
                Check every filtered column.
                */

                Object.keys(
                    selectedFilters
                ).forEach(columnIndex => {

                    if (!showRow) {
                        return;
                    }


                    const cell =
                        row.cells[columnIndex];

                    if (!cell) {
                        return;
                    }


                    const cellValue =
                        cell.textContent.trim();


                    /*
                    Multiple selections
                    within the same column use OR.

                    Example:

                    Status:
                    ☑ Active
                    ☑ Inactive

                    means Active OR Inactive.
                    */

                    const matches =
                        selectedFilters[columnIndex]
                            .includes(cellValue);


                    if (!matches) {

                        showRow = false;

                    }

                });


                if (showRow) {

                    row.style.display = "";

                    visibleRows++;

                } else {

                    row.style.display = "none";

                }

            });


            /*
            Show "No results"
            */

            if (visibleRows === 0) {

                noResults.style.display =
                    "block";

            } else {

                noResults.style.display =
                    "none";

            }


            /*
            Close menu
            */

            filterMenu.classList.remove(
                "show"
            );

        }


        /*
        ========================================
        CLEAR FILTERS
        ========================================
        */

        function clearFilters() {

            /*
            Uncheck everything
            */

            const checkboxes =
                filterOptions.querySelectorAll(
                    "input[type='checkbox']"
                );


            checkboxes.forEach(checkbox => {

                checkbox.checked = false;

            });


            /*
            Show every row
            */

            const rows =
                table.querySelectorAll(
                    "tbody tr"
                );


            rows.forEach(row => {

                row.style.display = "";

            });


            noResults.style.display =
                "none";

        }


        /*
        ========================================
        CREATE FILTERS WHEN PAGE LOADS
        ========================================
        */

        createFilters();

    </script>
