    <script>
        let customers = [];


        async function loadCustomers() {

            const message = document.getElementById("message");
            const customerList = document.getElementById("customerList");

            try {

                const response = await fetch("databaseview.php");

                const data = await response.json();

                if (!data.success) {

                    message.textContent = data.message;
                    return;

                }

                customers = data.data;

                displayCustomers(customers);

                message.textContent = "";

            } catch (error) {

                console.error(error);

                message.textContent = "Something went wrong.";

            }

        }


        function displayCustomers(customerData) {

            const customerList =
                document.getElementById("customerList");

            customerList.innerHTML = "";

            customerData.forEach(function(customer) {

                const row = document.createElement("tr");

                row.innerHTML = `
                    <td>${customer.cust_name}</td>

                    <td>${customer.cust_phone}</td>

                    <td>${customer.outlet}</td>

                     <td>${customer.attendee}</td>

                     <td>${customer.model}</td>

                     <td>${customer.imei}</td>

                     <td>${customer.purchase_date}</td>

                `;

                customerList.appendChild(row);

            });

        }


        function filteroutlet(broughtAt) {
            const outlet = broughtAt || document.getElementById("outletfilter").value;

    const table = document.getElementById("Data Table");
            const rows = table.getElementsByTagName("tr");
            rows.forEach(row => {
                const rowOutlet = row.cells[3].textContent;
                if (outlet === "ALL" || rowOutlet === outlet) {
                    row.style.display = "";
                } else {
                    row.style.display = "none";
                }
            });
            document.getElementById("outletfilter").addEventListener("change", function() {
                const selectedOutlet = this.value;
                filteroutlet(selectedOutlet);
            });
        }
       

        function filterCustomers() {

            const search =
                document.getElementById("search")
                .value
                .toLowerCase();

            const filteredCustomers = customers.filter(function(customer) {

                return (
                    customer.cust_name.toLowerCase().includes(search) ||
                    customer.cust_phone.includes(search)
                );

            });

            displayCustomers(filteredCustomers);

        }


        loadCustomers();


    </script>
